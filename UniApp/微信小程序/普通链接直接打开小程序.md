# 普通链接直接打开小程序

允许扫方形二维码打开小程序

# 在小程序后台配置

`开发管理-开发设置-扫普通链接二维码打开小程序`

该功能仅仅是劫持了微信的扫一扫功能，匹配链接则跳转到小程序指定页面，链接是否真的能访问，并不重要

# 测试

实际发布前，可以测试效果，对接参数

## “测试链接”指的是什么

指定一个链接（一般带特定参数），如果扫码结果就是这个链接，则让管理员/开发者进入小程序内指定页面

## “开发版”指的是什么

启动真机调试，上传完成后，手机扫码打开的那个就是“开发版”

# 读取参数的代码示例

```js
onLoad(options) {
	if (options.scene) {
		options = util.getUrlParams(decodeURIComponent(options.scene));
	}
	let {
		store_id,
		qrcode_id
	} = options;
	
	const q = decodeURIComponent(options.q)
	console.log("onLoad 的 q：", q)
	if(q && typeof q == 'string'){
		const queryPart = q.split('?')[1]
		console.log("onLoad 的 queryPart：", queryPart)
		if(queryPart){
			// “扫描普通二维码直接打开小程序”功能进入的
			const qObj = util.getUrlParams(queryPart)
			console.log("onLoad 的 qObj：", qObj)
			if(qObj.store_id){
				store_id = qObj.store_id
			}
			if(qObj.qrcode_id){
				qrcode_id = qObj.qrcode_id
			}
		}
	}

	this.info = {
		...this.info,
		store_id,
		qrcode_id
	};
}
```

```js
export default {
	getUrlParams: function(param, k, p) {
		if (typeof param != 'string') return {};
		k = k ? k : '&'; //整体参数分隔符
		p = p ? p : '='; //单个参数分隔符
		var value = {};
		if (param.indexOf(k) !== -1) {
			param = param.split(k);
			for (var val in param) {
				if (param[val].indexOf(p) !== -1) {
					var item = param[val].split(p);
					value[item[0]] = item[1];
				}
			}
		} else if (param.indexOf(p) !== -1) {
			var item = param.split(p);
			value[item[0]] = item[1];
		} else {
			return param;
		}
		return value;
	}
}
```
