# 施加力

```cs
	public override void _PhysicsProcess(double delta)
	{
		Vector2 localForce = Vector2.Zero;

		if (Input.IsActionPressed("go_forward"))
			localForce.Y -= MovementForce;
		if (Input.IsActionPressed("go_backward"))
			localForce.Y += MovementForce;
		if (Input.IsActionPressed("go_left"))
			localForce.X -= MovementForce;
		if (Input.IsActionPressed("go_right"))
			localForce.X += MovementForce;

		// 将本地坐标系下的力转换为世界坐标系
		Vector2 globalForce = Transform.X * localForce.X + Transform.Y * localForce.Y;
		ApplyCentralForce(globalForce);

		// 更新发动机喷射效果
		UpdateEnginePlume(localForce);

		// 阻力：与速度方向相反的力，大小正比于速度
		ApplyCentralForce(-DragCoefficient * LinearVelocity);

		// 旋转力矩
		if (Input.IsActionPressed("turn_left"))
			ApplyTorque(-RotationTorque);
		if (Input.IsActionPressed("turn_right"))
			ApplyTorque(RotationTorque);
	}
```
